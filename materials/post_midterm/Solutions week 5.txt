Exercise 1

import pandas as pd

def main(df):
    cols_to_use = []
    for col in df.columns:
        if col.islower():
            cols_to_use.append(col)
    return df[cols_to_use]

Exercise 2

import pandas as pd

def main(df):
    return df.loc[[row for row in df.index if row.month==4]].sort_index()

# Alternative solution (without list comprehension):

import pandas as pd

def main(df):
    rows = []
    for row in df.index:
        if row.month==4:
            rows.append(row)
    return df.loc[rows].sort_index()

import pandas as pd

def main(df):
    df["Weight (formatted)"] = df["Weight"].map(
        lambda x: f"{x * 1000:.0f} g")
    return df

import pandas as pd

def main(address, province):
    ext_address = address.merge(province, on="Town", how="left")
    return ext_address.fillna("No province")
